1. 系統技術架構與選型 (Architecture & Technology Stack)

為了實現「極致資安」、「即開即用」與「無伺服器相依性」的財務工具定位，本系統採取純前端無伺服器架構。

1.1 前端核心

• 網頁骨架 (HTML5)：採用語意化標籤，確保無障礙存取與對帳報表的可讀性。

• 樣式渲染 (CSS3)：引入 Tailwind CSS (via CDN)。利用原子化樣式快速構建響應式介面，並在匯出圖片時提供高保真的列印/對帳美感。

• 邏輯核心 (Native ES6+ JavaScript)：

	• 使用純原生 JavaScript 進行 DOM 操作與核心演算法撰寫。

	• 嚴禁使用任何打包工具（如 Webpack/Vite）或重型框架（如 React/Vue/Angular），保持單一 index.html 即可運行的超輕量特型。

• 截圖套件：引入 html2canvas (v1.4.1 或最新穩定版 CDN)。在純本地端沙盒中，將 HTML DOM 節點渲染為 Canvas，並導出為 PNG 格式。

1.2 零外部請求限制 (Strict Security Jail)

為了保護極敏感的薪資數據，開發必須嚴格遵守以下安全規範：

• 網頁中不得包含任何 fetch()、XMLHttpRequest、Image().src、或 navigator.sendBeacon 的外發業務請求。

• 載入的第三方套件（如 html2canvas 與 Tailwind CSS）必須採用預先驗證的官方 CDN (如 cdnjs 或 jsdelivr)，並附帶 integrity 屬性。

• 例外說明：專案內嵌 Google Analytics (GA) 以進行不具名之基本流量統計，該請求僅用於分析頁面造訪與功能點擊，絕不收集或傳輸任何使用者輸入之敏感財務資料。

3. 核心模組設計與程式實作細節 (Core Modules)

3.1 智慧解析器 (Parser Module)

智慧解析器負責將 Textarea 中的多行文字拆解，同時精準追蹤行號以應對輸入錯誤。

[javascript]
/**
 * 智慧解析器：將多行輸入解析為結構化 JSON 資料
 * @param {string} rawText 原始輸入文字
 * @returns {Array<{lineNo: number, name: string, amount: number}>} 解析後的資料陣列
 * @throws {Error} 當某行格式不符時，拋出帶有 lineNo 的錯誤
 */
function parseInput(rawText) {
  const lines = rawText.split('\n');
  const parsedRecords = [];

  for (let i = 0; i < lines.length; i++) {
    const lineNo = i + 1;
    const line = lines[i].trim();

    // 1. 自動過濾空白行
    if (line === '') {
      continue;
    }

    // 2. 識別分隔符：優先級為 Tab -> 半形逗號 -> 全形逗號
    let delimiter = null;
    if (line.includes('\t')) {
      delimiter = '\t';
    } else if (line.includes(',')) {
      delimiter = ',';
    } else if (line.includes('，')) {
      delimiter = '，';
    }

    if (!delimiter) {
      const err = new Error(`第 ${lineNo} 行格式錯誤：未偵測到有效的分隔符號 (Tab 或 逗號)`);
      err.lineNo = lineNo;
      throw err;
    }

    const parts = line.split(delimiter);
    if (parts.length !== 2) {
      const err = new Error(`第 ${lineNo} 行格式錯誤：欄位數不正確，應為「姓名,金額」`);
      err.lineNo = lineNo;
      throw err;
    }

    const rawName = parts[0].trim();
    const rawAmount = parts[1].trim();

    if (rawName === '') {
      const err = new Error(`第 ${lineNo} 行格式錯誤：姓名欄位不可空白`);
      err.lineNo = lineNo;
      throw err;
    }

    // 3. 清洗金額：移除非數字、非小數點之多餘貨幣符號與千分位符號
    // 清洗正則：移除 $, ￥, €, £ 以及所有半形 , 與全形 ，
    const cleanedAmount = rawAmount.replace(/[$,￥€£,，\s]/g, '');
    const amount = parseFloat(cleanedAmount);

    if (isNaN(amount) || amount <= 0) {
      const err = new Error(`第 ${lineNo} 行格式錯誤：「${rawAmount}」無法轉換為有效的金額`);
      err.lineNo = lineNo;
      throw err;
    }

    parsedRecords.push({
      lineNo,
      name: rawName,
      amount: amount
    });
  }

  return parsedRecords;
}


3.2 姓名正規化與同名加總 (Aggregator Module)

正規化模組用於消除輸入時的微小字元差異（如全形半形、多重空格），防止重複發放。

[javascript]
/**
 * 全形字元轉半形字元工具
 * @param {string} str 原始字串
 * @returns {string} 轉換後的半形字串
 */
function toHalfWidth(str) {
  return str.replace(/[\uFF01-\uFF5E]/g, function(char) {
    return String.fromCharCode(char.charCodeAt(0) - 0xfee0);
  }).replace(/\u3000/g, ' '); // 全形空格轉半形
}

/**
 * 姓名正規化與同名合併加總
 * @param {Array<{name: string, amount: number}>} records 
 * @returns {Array<{name: string, totalAmount: number}>}
 */
function aggregateRecords(records) {
  const groupMap = new Map();

  for (const record of records) {
    // 1. 全形轉半形
    let normName = toHalfWidth(record.name);
    // 2. 去除控制字元與首尾空格
    normName = normName.trim();
    // 3. 將多個連續空格收縮為單一空格
    normName = normName.replace(/\s+/g, ' ');

    // 以全小寫與縮小空格後的字串作為 Hash Key，防範大小寫與空格不一致
    const uniqueKey = normName.toLowerCase();

    if (groupMap.has(uniqueKey)) {
      const existing = groupMap.get(uniqueKey);
      existing.totalAmount += record.amount;
    } else {
      groupMap.set(uniqueKey, {
        name: normName, // 保留「第一次出現」的正常大小寫格式作為顯示姓名
        totalAmount: record.amount
      });
    }
  }

  // 將 Map 轉回陣列，並依顯示姓名進行穩定排序
  return Array.from(groupMap.values()).sort((a, b) => {
    return a.name.localeCompare(b.name, 'zh-Hant');
  });
}


3.3 面額拆解演算法 (Greedy Denomination Breakdown)

本系統使用經典的貪婪演算法（Greedy Algorithm）來將合併後的金額拆解為最少鈔券。

[javascript]
// 新台幣法定常用鈔幣面額陣列（降序排列）
const TWD_DENOMINATIONS = [1000, 500, 100, 50, 10, 5, 1];

/**
 * 將特定金額拆解為各面額之數量
 * @param {number} amount 應領總金額
 * @returns {Object} 各面額對應數量的 Key-Value 對應表
 */
function breakdownAmount(amount) {
  let remaining = Math.round(amount); // 四捨五入至整數，確保無小數點殘留
  const result = {};

  for (const denom of TWD_DENOMINATIONS) {
    result[denom] = Math.floor(remaining / denom);
    remaining %= denom;
  }

  return result;
}


4. 數據驗證與雙重對帳邏輯 (Data Integrity)

4.1 雙重對帳驗證機制 (Double-Entry Verification)

在 UI 渲染運算結果前，系統必須執行以下數學公式校驗：

1. 計算原始輸入加總：


$$S_{input} = \sum_{i=1}^{n} A_{i}$$


其中 ‭$A_{i}$‬ 代表第 ‭$i$‬ 筆解析成功後的原始金額。

2. 計算面額拆解大計：


$$S_{breakdown} = \sum_{p \in P} \left( \sum_{d \in D} N_{p, d} \times d \right)$$


其中 ‭$P$‬ 為合併後的個人名單，‭$D$‬ 為面額集合 ‭$\{1000, 500, 100, 50, 10, 5, 1\}$‬，‭$N_{p, d}$‬ 為人員 ‭$p$‬ 在面額 ‭$d$‬ 拆解出的張數或個數。

3. 驗證與異常阻斷：
若 ‭$S_{input} \neq S_{breakdown}$‬，系統必須：

	• 拋出顯著的對帳失敗警告（紅色 Modal 或 Alert）。

	• 立刻將 document.getElementById('btn-export') 設為 disabled = true，完全杜絕錯誤數據報表產出。

4.2 錯誤行自動選取與高亮實作 (Error Line Auto-Selection)

當 parseInput() 偵測到錯誤並拋出異常時，UI 控制模組必須精準定位該行：

[javascript]
function handleErrorHighlight(rawText, errorLineNo) {
  const textarea = document.getElementById('input-textarea');
  const lines = rawText.split('\n');
  
  let startOffset = 0;
  let endOffset = 0;

  for (let i = 0; i < lines.length; i++) {
    if (i === errorLineNo - 1) {
      endOffset = startOffset + lines[i].length;
      break;
    }
    // 加上換行符號 \n 的長度
    startOffset += lines[i].length + 1;
  }

  // 自動聚焦輸入框並反白出錯行
  textarea.focus();
  textarea.setSelectionRange(startOffset, endOffset);
}

5.2 匯出純淨版面控制邏輯 (Clean Canvas Exporting)

為了在對帳單上隱藏按鈕元件，匯出控制函式必須按照以下生命週期執行：

[javascript]
async function exportReportAsImage() {
  const reportContainer = document.getElementById('report-container');
  const actionButtons = document.querySelectorAll('.no-export'); // 需要隱藏的按鈕加上此 Class

  // 1. 隱藏操作按鈕
  actionButtons.forEach(btn => btn.style.setProperty('display', 'none', 'important'));

  // 2. 觸發快門特效
  reportContainer.classList.add('flash-active');

  try {
    // 3. 呼叫 html2canvas
    const canvas = await html2canvas(reportContainer, {
      scale: 2, // 提高 2 倍解析度，確保列印與縮放不模糊
      useCORS: true,
      backgroundColor: '#ffffff', // 強制白色背景
      logging: false
    });

    // 4. 轉成圖片並自動下載
    const imageURI = canvas.toDataURL('image/png');
    const link = document.createElement('a');
    link.download = `MoneySnap_兌幣明細表_${new Date().toISOString().slice(0, 10)}.png`;
    link.href = imageURI;
    link.click();
  } catch (error) {
    console.error('匯出圖片時發生錯誤:', error);
  } finally {
    // 5. 復原按鈕與特效
    setTimeout(() => {
      actionButtons.forEach(btn => btn.style.removeProperty('display'));
      reportContainer.classList.remove('flash-active');
    }, 300);
  }
}
